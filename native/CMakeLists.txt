cmake_minimum_required(VERSION 3.16)
project(edge_rkllm LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "Build this adapter on ARM64 Linux, or use an explicit ARM64 Linux toolchain file.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
foreach(NAME rkllmrt rknnrt)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/lib/lib${NAME}.so")
    message(FATAL_ERROR "Missing lib${NAME}.so. Run scripts/setup_assets.py --sdk first.")
  endif()
  add_library(${NAME} SHARED IMPORTED)
  set_target_properties(${NAME} PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/lib/lib${NAME}.so" IMPORTED_NO_SONAME TRUE)
endforeach()
find_package(Threads REQUIRED)
add_executable(edge-rkllm edge_rkllm.cpp encoder_safe.cpp)
target_compile_options(edge-rkllm PRIVATE -Wall -Wextra)
target_link_libraries(edge-rkllm PRIVATE rkllmrt rknnrt Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(edge-rkllm PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
add_custom_command(TARGET edge-rkllm POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:edge-rkllm>/lib"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/lib/librkllmrt.so" "$<TARGET_FILE_DIR:edge-rkllm>/lib/"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/lib/librknnrt.so" "$<TARGET_FILE_DIR:edge-rkllm>/lib/")
install(TARGETS edge-rkllm RUNTIME DESTINATION .)
install(FILES lib/librkllmrt.so lib/librknnrt.so DESTINATION lib)
